// ─────────────────────────────────────────────────────────────────────────────
//  XAU Volume Bubble + VWAP + ORB
//
//  • Volume Bubble : statistically unusual volume (z-score, optional time-of-day
//                    baseline), colored by ESTIMATED buy/sell delta.
//  • VWAP          : anchored (Session / Week / Month) with ±σ bands.
//  • ORB           : Opening Range (NY / COMEX / London / Asia / Custom) with
//                    high/low/mid lines, extension targets, breakout and
//                    failed-breakout (trap) markers.
//
//  IMPORTANT
//  • On XAUUSD CFD feeds, "volume" is broker TICK COUNT (activity), not traded
//    size. Delta here is an ESTIMATE from lower-timeframe candle direction
//    (close vs open + tick rule), the same heuristic as TradingView's
//    aggregated Volume Delta. It is not exchange aggressor data.
//  • All thresholds are proposed defaults, NOT validated or backtested.
//  • Signals are evaluated on confirmed (closed) bars only.
// ─────────────────────────────────────────────────────────────────────────────
indicator("XAU Volume Bubble + VWAP + ORB", shorttitle = "VB·VWAP·ORB", overlay = true,
     max_labels_count = 500, max_lines_count = 500, max_boxes_count = 200)

// ═════════════════════════════ INPUTS ═══════════════════════════════════════

const string G_VW   = "VWAP"
const string G_ORB  = "Opening Range (ORB)"
const string G_SIG  = "Signals"
const string G_UI   = "Style & Dashboard"

bool   showBub   = input.bool(true, "Show volume bubbles", group = G_VB)
string ltfMode   = input.string("Auto", "Intrabar timeframe (delta estimate)", options = ["Auto", "1", "5", "15", "60"], group = G_VB,
     tooltip = "Auto: chart ≤15m → 1m, ≤4h → 5m, otherwise 60m. Lower = more precise but shorter history.")
int    zLen      = input.int(60, "Volume z-score lookback", minval = 10, group = G_VB)
float  zThr      = input.float(2.0, "Bubble threshold (z)", minval = 0.5, step = 0.1, group = G_VB,
     tooltip = "Tier 1 ≥ z, Tier 2 ≥ z+1, Tier 3 ≥ z+2")
bool   useTod    = input.bool(true, "Time-of-day volume baseline", group = G_VB,
     tooltip = "Compare each bar's volume with the same clock-time slot of previous days, so the NY session does not bubble every bar.")
int    todDays   = input.int(20, "Days in time-of-day baseline", minval = 5, maxval = 200, group = G_VB)
string bubPos    = input.string("Intrabar VWAP", "Bubble position", options = ["Intrabar VWAP", "Close", "HLC3"], group = G_VB)
float  absBody   = input.float(0.3, "Absorption: max body (× ATR)", minval = 0.05, step = 0.05, group = G_VB,
     tooltip = "A bubble whose candle body is smaller than this × ATR(14) is marked as absorption (big effort, little result).")
bool   glow      = input.bool(true, "Glow effect", group = G_VB)

bool   showVwap  = input.bool(true, "Show VWAP", group = G_VW)

bool   showBands = input.bool(true, "Show bands", group = G_VW)
float  band1     = input.float(1.0, "Band 1 (σ)", minval = 0.1, step = 0.1, group = G_VW)
float  band2     = input.float(2.0, "Band 2 (σ)", minval = 0.1, step = 0.1, group = G_VW)

bool   showOrb   = input.bool(true, "Show ORB", group = G_ORB)
string orbPreset = input.string("New York 09:30 ET", "Opening session",
     options = ["New York 09:30 ET", "COMEX 08:20 ET", "London 08:00 UK", "Asia 09:00 Tokyo", "Custom"], group = G_ORB)
int    orbLen    = input.int(30, "Range length (minutes)", options = [5, 15, 30, 60, 120, 240], group = G_ORB)
int    custH     = input.int(9, "Custom start hour", minval = 0, maxval = 23, inline = "cust", group = G_ORB)
int    custM     = input.int(30, "min", minval = 0, maxval = 59, inline = "cust", group = G_ORB)
string custTz    = input.string("America/New_York", "Custom timezone", group = G_ORB)
int    holdHrs   = input.int(8, "Extend lines (hours after range)", minval = 1, maxval = 24, group = G_ORB)
bool   showExt   = input.bool(true, "Show extension targets (1R / 2R)", group = G_ORB)
bool   fillZone  = input.bool(true, "Fill range zone", group = G_ORB)

bool   showBrk   = input.bool(true, "Show ORB breakout markers", group = G_SIG)
int    failBars  = input.int(3, "Failed breakout window (bars)", minval = 1, maxval = 20, group = G_SIG,
     tooltip = "If price closes back inside the range within N bars after a breakout, it is marked as a failed breakout (possible trapped traders).")

color  cBuy      = input.color(#00E5A0, "Buy", inline = "c1", group = G_UI)
color  cSell     = input.color(#FF4D6D, "Sell", inline = "c1", group = G_UI)
color  cAbs      = input.color(#FFC93C, "Absorption", inline = "c1", group = G_UI)
color  cVwap     = input.color(#B388FF, "VWAP", inline = "c2", group = G_UI)
color  cOrH      = input.color(#29B6F6, "OR High", inline = "c3", group = G_UI)
color  cOrL      = input.color(#FF8A65, "OR Low", inline = "c3", group = G_UI)
color  cOrMid    = input.color(#9E9E9E, "Mid", inline = "c3", group = G_UI)
bool   showDash  = input.bool(true, "Show dashboard", group = G_UI)

// ═════════════════════════════ HELPERS ══════════════════════════════════════
f_tfLabel(string tf) =>
    float m = str.tonumber(tf)
    na(m) ? tf : m >= 60 ? str.tostring(m / 60) + "h" : str.tostring(m) + "m"

f_clamp(float x, float lo, float hi) =>
    math.max(lo, math.min(hi, x))

// ═════════════════════════════ INTRABAR DATA ════════════════════════════════
int    chartSec = timeframe.in_seconds()
string autoLtf  = chartSec <= 900 ? "1" : chartSec <= 14400 ? "5" : "60"
string ltf      = ltfMode == "Auto" ? autoLtf : ltfMode

[ibO, ibC, ibV, ibH, ibL, ibT] = request.security_lower_tf(syminfo.tickerid, ltf, [open, close, volume, high, low, time],
     ignore_invalid_timeframe = true)

int ibN = na(ibV) ? 0 : ibV.size()

// ── Delta estimate: intrabar polarity + tick rule (state carried across bars)
var float lastSign = 1.0
var float prevIbC  = na
float buyV  = 0.0
float sellV = 0.0
float pvSum = 0.0
float ibVol = 0.0

if ibN > 0
    for i = 0 to ibN - 1
        float o = ibO.get(i)
        float c = ibC.get(i)
        float v = nz(ibV.get(i))
        float s = c > o ? 1.0 : c < o ? -1.0 : na(prevIbC) ? lastSign : c > prevIbC ? 1.0 : c < prevIbC ? -1.0 : lastSign
        lastSign := s
        prevIbC  := c
        if s > 0
            buyV += v
        else
            sellV += v
        pvSum += v * (ibH.get(i) + ibL.get(i) + c) / 3.0
        ibVol += v
else
    // Fallback: bar-shape split (volume × CLV). A rough estimate only.
    float rngFb = high - low
    float vFb   = nz(volume)
    float dFb   = rngFb > 0 ? vFb * (2 * close - high - low) / rngFb : 0.0
    buyV  := (vFb + dFb) / 2
    sellV := (vFb - dFb) / 2

bool  usedFallback = ibN == 0
float totV     = buyV + sellV
float delta    = buyV - sellV
float deltaPct = totV > 0 ? 100 * delta / totV : 0.0
float ibVwap   = ibVol > 0 ? pvSum / ibVol : hlc3

// ═════════════════════════════ VOLUME BUBBLE ════════════════════════════════
type SlotBuf
    array<float> vals

var map<int, SlotBuf> slots = map.new<int, SlotBuf>()
float slotMean = na
float slotSd   = na
if useTod
    int slotKey = hour(time, "UTC") * 60 + minute(time, "UTC")
    if not slots.contains(slotKey)
        slots.put(slotKey, SlotBuf.new(array.new<float>()))
    SlotBuf sb = slots.get(slotKey)
    if sb.vals.size() >= 5
        slotMean := sb.vals.avg()
        slotSd   := sb.vals.stdev()
    if barstate.isconfirmed
        sb.vals.push(nz(volume))
        if sb.vals.size() > todDays
            sb.vals.shift()

float vMean = ta.sma(volume, zLen)[1]   // baseline excludes the current bar
float vSd   = ta.stdev(volume, zLen)[1]
float zRoll = vSd > 0 ? (volume - vMean) / vSd : 0.0
bool  todOk = useTod and not na(slotSd) and slotSd > 0
float zVol  = todOk ? (volume - slotMean) / slotSd : zRoll

int   tier   = zVol >= zThr + 2 ? 3 : zVol >= zThr + 1 ? 2 : zVol >= zThr ? 1 : 0
float atr    = ta.atr(14)
bool  absorb = tier > 0 and math.abs(close - open) <= absBody * atr
float bubY   = bubPos == "Close" ? close : bubPos == "HLC3" ? hlc3 : ibVwap

if showBub and tier > 0 and barstate.isconfirmed
    color  base   = absorb ? cAbs : delta >= 0 ? cBuy : cSell
    int    transp = int(f_clamp(55 - math.abs(deltaPct) * 0.5, 10, 55))
    string szCore = tier == 3 ? size.large : tier == 2 ? size.normal : size.small
    string szGlow = tier == 3 ? size.huge : tier == 2 ? size.large : size.normal
    string kind   = absorb ? "Absorption (big effort, small body)" : delta >= 0 ? "Buy-dominant (estimated)" : "Sell-dominant (estimated)"
    string src    = usedFallback ? "bar-shape estimate" : "intrabar " + f_tfLabel(ltf)
    string tip    = kind + "\nVolume: " + str.tostring(volume, format.volume) + "\nz-score: " + str.tostring(zVol, "#.00") +
         "\nBuy est.: " + str.tostring(buyV, format.volume) + "\nSell est.: " + str.tostring(sellV, format.volume) +
         "\nΔ%: " + str.tostring(deltaPct, "#.0") + "%\nDelta source: " + src
    if glow
        label.new(bar_index, bubY, "", color = color.new(base, 85), style = label.style_circle, size = szGlow)
    label.new(bar_index, bubY, "", color = color.new(base, transp), style = label.style_circle, size = szCore, tooltip = tip)

// ═════════════════════════════ VWAP ═════════════════════════════════════════
bool chD = timeframe.change("D")
bool chW = timeframe.change("W")
bool chM = timeframe.change("M")
bool newAnchor = vwAnchor == "Session" ? chD : vwAnchor == "Week" ? chW : chM

var float sPV  = 0.0
var float sV   = 0.0
var float sPV2 = 0.0
if newAnchor
    sPV  := 0.0
    sV   := 0.0
    sPV2 := 0.0
float vSrc = hlc3
float vVol = nz(volume)
sPV  += vSrc * vVol
sV   += vVol
sPV2 += vSrc * vSrc * vVol
int   anchorSec = vwAnchor == "Session" ? 86400 : vwAnchor == "Week" ? 604800 : 2419200
bool  vwapOk  = chartSec < anchorSec
float vwapVal = vwapOk and sV > 0 ? sPV / sV : na
float vwapSd  = vwapOk and sV > 0 ? math.sqrt(math.max(sPV2 / sV - vwapVal * vwapVal, 0.0)) : na

color cLine  = newAnchor ? na : cVwap
color cBand1 = newAnchor ? na : color.new(cVwap, 55)
color cBand2 = newAnchor ? na : color.new(cVwap, 70)
pVw = plot(showVwap ? vwapVal : na, "VWAP", color = cLine, linewidth = 2)
pU1 = plot(showVwap and showBands ? vwapVal + band1 * vwapSd : na, "VWAP +1", color = cBand1)
pU2 = plot(showVwap and showBands ? vwapVal + band2 * vwapSd : na, "VWAP +2", color = cBand2)
pL1 = plot(showVwap and showBands ? vwapVal - band1 * vwapSd : na, "VWAP −1", color = cBand1)
pL2 = plot(showVwap and showBands ? vwapVal - band2 * vwapSd : na, "VWAP −2", color = cBand2)
fill(pU1, pU2, color = newAnchor ? na : color.new(cSell, 90), title = "Upper band zone")
fill(pL1, pL2, color = newAnchor ? na : color.new(cBuy, 90), title = "Lower band zone")

// ═════════════════════════════ OPENING RANGE ════════════════════════════════
int orStartMin = switch orbPreset
    "New York 09:30 ET" => 570
    "COMEX 08:20 ET"    => 500
    "London 08:00 UK"   => 480
    "Asia 09:00 Tokyo"  => 540
    => custH * 60 + custM
string orTz = switch orbPreset
    "New York 09:30 ET" => "America/New_York"
    "COMEX 08:20 ET"    => "America/New_York"
    "London 08:00 UK"   => "Europe/London"
    "Asia 09:00 Tokyo"  => "Asia/Tokyo"
    => custTz

var float orH        = na
var float orL        = na
var int   orDay      = na
var int   orStartTs  = na
var int   orEndTs    = na
var bool  orBuilding = false
var bool  orDone     = false
var bool  orApprox   = false
bool newOr  = false
bool justDone = false

// Each "unit" is one intrabar (or the chart bar itself when no intrabars exist).
// A unit is used when it OVERLAPS the window. Windows may cross midnight.
// If a unit sticks out of the window, the range is flagged as approximate.
bool chartOk = chartSec <= orbLen * 60
int  nUnits  = ibN > 0 ? ibN : chartOk ? 1 : 0
int  uLen    = ibN > 0 ? math.max(1, int(timeframe.in_seconds(ltf) / 60)) : math.max(1, int(chartSec / 60))
if nUnits > 0
    for i = 0 to nUnits - 1
        int   t   = ibN > 0 ? ibT.get(i) : time
        float uh  = ibN > 0 ? ibH.get(i) : high
        float ul  = ibN > 0 ? ibL.get(i) : low
        int   md  = hour(t, orTz) * 60 + minute(t, orTz)
        int   rel = (md - orStartMin + 1440) % 1440          // minutes since window start
        bool  inside = rel < orbLen
        bool  before = not inside and rel > 1440 - uLen       // starts before, ends inside
        if inside or before
            int  wStart = inside ? t - rel * 60000 : t + (1440 - rel) * 60000
            int  dk     = year(wStart, orTz) * 10000 + month(wStart, orTz) * 100 + dayofmonth(wStart, orTz)
            bool sticks = before or rel + uLen > orbLen
            if na(orDay) or dk != orDay
                orDay      := dk
                orH        := uh
                orL        := ul
                orStartTs  := wStart
                orEndTs    := wStart + orbLen * 60000
                orBuilding := true
                orDone     := false
                orApprox   := sticks
                newOr      := true
            else if orBuilding
                orH := math.max(orH, uh)
                orL := math.min(orL, ul)
                orApprox := orApprox or sticks

if orBuilding and not na(orEndTs) and time_close >= orEndTs
    orBuilding := false
    orDone     := true
    justDone   := true

bool  orActive = not na(orEndTs) and time < orEndTs + holdHrs * 3600000
float orRng    = orH - orL

// ── Drawings
var line  lnH   = na
var line  lnL   = na
var line  lnM   = na
var box   bxZ   = na
var label tgH   = na
var label tgL   = na
var array<line>  extLn = array.new<line>()
var array<label> extTg = array.new<label>()
color noCol = color.new(color.black, 100)

if showOrb and newOr
    extLn.clear()
    extTg.clear()
    lnH := line.new(bar_index, orH, bar_index, orH, color = cOrH, width = 2, style = line.style_dashed)
    lnL := line.new(bar_index, orL, bar_index, orL, color = cOrL, width = 2, style = line.style_dashed)
    lnM := line.new(bar_index, (orH + orL) / 2, bar_index, (orH + orL) / 2, color = color.new(cOrMid, 30), width = 1, style = line.style_dotted)
    bxZ := box.new(bar_index, orH, bar_index, orL, border_color = noCol,
         bgcolor = fillZone ? color.new(cOrH, 90) : noCol)
    tgH := label.new(bar_index, orH, "ORH", style = label.style_label_left, color = noCol, textcolor = cOrH, size = size.small)
    tgL := label.new(bar_index, orL, "ORL", style = label.style_label_left, color = noCol, textcolor = cOrL, size = size.small)

if showOrb and not na(lnH) and orActive
    float mid = (orH + orL) / 2
    lnH.set_xy2(bar_index, orH)
    lnH.set_y1(orH)
    lnL.set_xy2(bar_index, orL)
    lnL.set_y1(orL)
    lnM.set_xy2(bar_index, mid)
    lnM.set_y1(mid)
    bxZ.set_top(orH)
    bxZ.set_bottom(orL)
    if orBuilding or justDone
        bxZ.set_right(bar_index)
    tgH.set_xy(bar_index, orH)
    tgH.set_text("ORH  " + str.tostring(orH, format.mintick))
    tgL.set_xy(bar_index, orL)
    tgL.set_text("ORL  " + str.tostring(orL, format.mintick))
    for ln in extLn
        ln.set_x2(bar_index)
    for tg in extTg
        tg.set_x(bar_index)

if showOrb and justDone and not na(lnH)
    lnH.set_style(line.style_solid)
    lnL.set_style(line.style_solid)
    bxZ.set_bgcolor(fillZone ? color.new(cOrH, 92) : noCol)
    if showExt and orRng > 0
        for k = 1 to 2
            float up = orH + k * orRng
            float dn = orL - k * orRng
            extLn.push(line.new(bar_index, up, bar_index, up, color = color.new(cOrH, 45), width = 1, style = line.style_dotted))
            extLn.push(line.new(bar_index, dn, bar_index, dn, color = color.new(cOrL, 45), width = 1, style = line.style_dotted))
            extTg.push(label.new(bar_index, up, "+" + str.tostring(k) + "R  " + str.tostring(up, format.mintick),
                 style = label.style_label_left, color = noCol, textcolor = color.new(cOrH, 30), size = size.tiny))
            extTg.push(label.new(bar_index, dn, "−" + str.tostring(k) + "R  " + str.tostring(dn, format.mintick),
                 style = label.style_label_left, color = noCol, textcolor = color.new(cOrL, 30), size = size.tiny))

// ═════════════════════════════ ORB SIGNALS ══════════════════════════════════
var bool brokeUp  = false
var bool brokeDn  = false
var bool failedUp = false
var bool failedDn = false
var int  brkUpBar = na
var int  brkDnBar = na
if newOr
    brokeUp  := false
    brokeDn  := false
    failedUp := false
    failedDn := false

bool brkUp  = false
bool brkDn  = false
bool failUp = false
bool failDn = false
if barstate.isconfirmed and orDone and orActive
    if brokeUp and not failedUp and bar_index > brkUpBar and bar_index - brkUpBar <= failBars and close < orH
        failedUp := true
        failUp   := true
    if brokeDn and not failedDn and bar_index > brkDnBar and bar_index - brkDnBar <= failBars and close > orL
        failedDn := true
        failDn   := true
    if not brokeUp and close > orH
        brokeUp  := true
        brkUp    := true
        brkUpBar := bar_index
    if not brokeDn and close < orL
        brokeDn  := true
        brkDn    := true
        brkDnBar := bar_index

bool aboveVw  = not na(vwapVal) and close > vwapVal
bool belowVw  = not na(vwapVal) and close < vwapVal
bool strongUp = brkUp and tier > 0 and delta > 0 and aboveVw
bool strongDn = brkDn and tier > 0 and delta < 0 and belowVw

if showBrk and brkUp
    label.new(bar_index, low, strongUp ? "ORB ▲ ★" : "ORB ▲", style = label.style_label_up, color = cBuy, textcolor = #0B0E14, size = size.small,
         tooltip = strongUp ? "Breakout above OR high with volume bubble, buy-dominant delta and price above VWAP" : "Close above OR high")
if showBrk and brkDn
    label.new(bar_index, high, strongDn ? "ORB ▼ ★" : "ORB ▼", style = label.style_label_down, color = cSell, textcolor = #0B0E14, size = size.small,
         tooltip = strongDn ? "Breakdown below OR low with volume bubble, sell-dominant delta and price below VWAP" : "Close below OR low")
if showBrk and failUp
    label.new(bar_index, high, "✕ Fail", style = label.style_label_down, color = cAbs, textcolor = #0B0E14, size = size.tiny,
         tooltip = "Breakout failed: closed back below OR high → possible trapped longs")
if showBrk and failDn
    label.new(bar_index, low, "✕ Fail", style = label.style_label_up, color = cAbs, textcolor = #0B0E14, size = size.tiny,
         tooltip = "Breakdown failed: closed back above OR low → possible trapped shorts")

// ═════════════════════════════ DATA WINDOW ══════════════════════════════════
plot(deltaPct, "Delta % (est.)", display = display.data_window)
plot(zVol, "Volume z", display = display.data_window)
plot(tier, "Bubble tier", display = display.data_window)

// ═════════════════════════════ DASHBOARD ════════════════════════════════════
var int nIb = 0
var int nFb = 0
if usedFallback
    nFb += 1
else
    nIb += 1

string dashPos = switch dashPosIn
    "Bottom right" => position.bottom_right
    "Top left"     => position.top_left
    "Bottom left"  => position.bottom_left
    => position.top_right

color cPanel = color.new(#0B0E14, 15)
color cText  = #D1D4DC
color cMuted = #787B86
var table dash = table.new(dashPos, 2, 10, bgcolor = cPanel, frame_color = color.new(cVwap, 60), frame_width = 1,
     border_color = color.new(#2A2E39, 0), border_width = 1)

f_row(table tb, int r, string k, string v, color vc) =>
    tb.cell(0, r, k, text_color = cMuted, text_size = size.small, text_halign = text.align_left)
    tb.cell(1, r, v, text_color = vc, text_size = size.small, text_halign = text.align_right)

if showDash and barstate.isfirst
    dash.cell(0, 0, "VB · VWAP · ORB", text_color = cVwap, text_size = size.normal)
    dash.merge_cells(0, 0, 1, 0)

if showDash and barstate.islast
    string srcTxt = usedFallback ? "Bar-shape est." : "Intrabar " + f_tfLabel(ltf)
    f_row(dash, 1, "Δ source", srcTxt, usedFallback ? cAbs : cText)
    f_row(dash, 2, "Bar Δ% (est.)", str.tostring(deltaPct, "#.0") + "%", delta >= 0 ? cBuy : cSell)
    f_row(dash, 3, "Volume z", str.tostring(zVol, "#.00") + (tier > 0 ? "  ●" : ""), tier > 0 ? (absorb ? cAbs : delta >= 0 ? cBuy : cSell) : cText)
    float sig = vwapSd > 0 ? (close - vwapVal) / vwapSd : 0.0
    f_row(dash, 4, "vs VWAP", not vwapOk ? "TF ≥ anchor" : na(vwapVal) ? "n/a" : (aboveVw ? "Above " : "Below ") + str.tostring(sig, "+#.0;-#.0") + "σ", aboveVw ? cBuy : cSell)
    string orState = na(orH) ? "No range yet" : orBuilding ? "Building…" : close > orH ? "Above range" : close < orL ? "Below range" : "Inside range"
    color  orCol   = na(orH) ? cMuted : orBuilding ? cAbs : close > orH ? cOrH : close < orL ? cOrL : cText
    f_row(dash, 5, "ORB", orState, orCol)
    f_row(dash, 6, "OR range", na(orRng) ? "n/a" : str.tostring(orRng, format.mintick) + (orApprox ? "  ≈" : ""), orApprox ? cAbs : cText)
    float cov = 100.0 * nIb / math.max(1, nIb + nFb)
    f_row(dash, 7, "Intrabar coverage", str.tostring(cov, "#") + "% of bars", cov < 50 ? cAbs : cText)
    bool tickVol = syminfo.type != "futures" and syminfo.type != "crypto" and syminfo.type != "stock"
    f_row(dash, 8, "Volume type", tickVol ? "Tick count (proxy)" : "Exchange volume", tickVol ? cAbs : cText)
    f_row(dash, 9, "Bubble baseline", todOk ? "Time-of-day" : "Rolling " + str.tostring(zLen), cText)
